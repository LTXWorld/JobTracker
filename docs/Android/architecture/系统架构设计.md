# JobView Android 系统架构设计

## 1. 执行概要

本文档定义了 JobView Android 移动应用的完整系统架构设计。基于 React Native 跨平台开发框架，采用现代化的前端架构模式，实现高性能、可扩展、易维护的求职投递记录管理移动应用。

### 2.2 核心架构原则

1. **单一职责原则**: 每个组件和模块都有明确的职责
2. **模块化设计**: 功能模块独立开发，松耦合高内聚
3. **状态管理集中化**: 使用 Redux Toolkit 统一管理应用状态
4. **数据流单向性**: 数据从 Store 流向组件，事件向上传播
5. **离线优先**: 设计支持离线使用的数据架构
6. **性能优化**: 采用懒加载、虚拟化等技术优化性能

```typescript
// hooks/useOfflineSync.ts
export const useOfflineSync = () => {
  const dispatch = useAppDispatch()
  const { isOnline, syncQueue } = useAppSelector(state => state.sync)

  useEffect(() => {
    if (isOnline && syncQueue.length > 0) {
      dispatch(processSyncQueue())
    }
  }, [isOnline, syncQueue.length, dispatch])

  const addToSyncQueue = useCallback((operation: SyncOperation) => {
    dispatch(addToQueue(operation))
  }, [dispatch])

  return {
    isOnline,
    queueSize: syncQueue.length,
    addToSyncQueue,
  }
}
```

### 6.3 数据同步策略

```typescript
// services/SyncService.ts
class SyncService {
  async syncToServer(): Promise<SyncResult> {
    const pendingApplications = await database.collections
      .get<Application>('applications')
      .query(Q.where('sync_status', 'pending'))
      .fetch()

    const syncOperations = []

    for (const app of pendingApplications) {
      if (app.serverCreatedAt) {
        // 更新操作
        syncOperations.push(this.updateOnServer(app))
      } else {
        // 创建操作
        syncOperations.push(this.createOnServer(app))
      }
    }

    const results = await Promise.allSettled(syncOperations)
    return this.processSyncResults(results)
  }

  async syncFromServer(lastSyncTime?: string): Promise<void> {
    const response = await api.get('/mobile/sync/incremental', {
      params: { last_sync: lastSyncTime }
    })

    const { applications, conflicts } = response.data

    if (conflicts.length > 0) {
      await this.handleConflicts(conflicts)
    }

    await database.batch(
      ...applications.created.map(app =>
        database.collections.get<Application>('applications')
          .prepareCreate(record => Object.assign(record, app))
      ),
      ...applications.updated.map(app =>
        this.updateLocalRecord(app)
      ),
      ...applications.deleted.map(id =>
        this.deleteLocalRecord(id)
      )
    )
  }

  private async handleConflicts(conflicts: SyncConflict[]): Promise<void> {
    for (const conflict of conflicts) {
      const resolution = await this.resolveConflict(conflict)
      await this.applyResolution(conflict, resolution)
    }
  }
}
```

```typescript
// services/AuthService.ts
class AuthService {
  private biometricService: BiometricService
  private tokenService: TokenService

  async login(credentials: LoginCredentials): Promise<AuthResult> {
    try {
      const response = await api.post('/auth/login', credentials)
      const { token, refreshToken, user } = response.data

      await this.tokenService.storeTokens(token, refreshToken)
      await this.storeUserData(user)

      return { success: true, user, token }
    } catch (error) {
      throw new AuthError('Login failed', error)
    }
  }

  async enableBiometric(): Promise<void> {
    const isSupported = await this.biometricService.isSupported()
    if (!isSupported) {
      throw new Error('Biometric authentication not supported')
    }

    const isEnrolled = await this.biometricService.isEnrolled()
    if (!isEnrolled) {
      throw new Error('No biometric data enrolled')
    }

    await this.biometricService.createKey('auth_key')
    await AsyncStorage.setItem('biometric_enabled', 'true')
  }

  async biometricLogin(): Promise<AuthResult> {
    const encryptedData = await AsyncStorage.getItem('encrypted_auth_data')
    if (!encryptedData) {
      throw new Error('No biometric login data found')
    }

    const decryptedData = await this.biometricService.decrypt(
      'auth_key',
      encryptedData
    )

    return this.validateStoredAuth(JSON.parse(decryptedData))
  }
}

// services/TokenService.ts
class TokenService {
  async storeTokens(token: string, refreshToken: string): Promise<void> {
    const encryptedToken = await encrypt(token)
    const encryptedRefreshToken = await encrypt(refreshToken)

    await AsyncStorage.multiSet([
      ['auth_token', encryptedToken],
      ['refresh_token', encryptedRefreshToken],
    ])
  }

  async getValidToken(): Promise<string | null> {
    const encryptedToken = await AsyncStorage.getItem('auth_token')
    if (!encryptedToken) return null

    const token = await decrypt(encryptedToken)

    if (this.isTokenExpired(token)) {
      return await this.refreshToken()
    }

    return token
  }

  private async refreshToken(): Promise<string | null> {
    const refreshToken = await this.getRefreshToken()
    if (!refreshToken) return null

    try {
      const response = await api.post('/auth/refresh', { refreshToken })
      const { token: newToken, refreshToken: newRefreshToken } = response.data

      await this.storeTokens(newToken, newRefreshToken)
      return newToken
    } catch (error) {
      await this.clearTokens()
      throw new AuthError('Token refresh failed')
    }
  }
}
```

### 7.2 数据加密架构

```typescript
// services/EncryptionService.ts
class EncryptionService {
  private keychain = new Keychain()

  async generateKey(keyAlias: string): Promise<void> {
    await this.keychain.setInternetCredentials(
      keyAlias,
      keyAlias,
      await this.generateSecureKey()
    )
  }

  async encrypt(keyAlias: string, data: string): Promise<string> {
    const key = await this.keychain.getInternetCredentials(keyAlias)
    if (!key) throw new Error('Encryption key not found')

    const encrypted = CryptoJS.AES.encrypt(data, key.password).toString()
    return encrypted
  }

  async decrypt(keyAlias: string, encryptedData: string): Promise<string> {
    const key = await this.keychain.getInternetCredentials(keyAlias)
    if (!key) throw new Error('Decryption key not found')

    const decrypted = CryptoJS.AES.decrypt(encryptedData, key.password)
    return decrypted.toString(CryptoJS.enc.Utf8)
  }

  private async generateSecureKey(): Promise<string> {
    return CryptoJS.lib.WordArray.random(256/8).toString()
  }
}
```

```typescript
// services/MemoryService.ts
class MemoryService {
  private imageCache = new Map<string, string>()
  private maxCacheSize = 50 * 1024 * 1024 // 50MB
  private currentCacheSize = 0

  async cacheImage(url: string): Promise<string> {
    if (this.imageCache.has(url)) {
      return this.imageCache.get(url)!
    }

    try {
      const localPath = await this.downloadAndCacheImage(url)
      const imageSize = await this.getImageSize(localPath)

      if (this.currentCacheSize + imageSize > this.maxCacheSize) {
        await this.clearOldestImages(imageSize)
      }

      this.imageCache.set(url, localPath)
      this.currentCacheSize += imageSize

      return localPath
    } catch (error) {
      console.warn('Failed to cache image:', error)
      return url
    }
  }

  private async clearOldestImages(requiredSize: number): Promise<void> {
    const entries = Array.from(this.imageCache.entries())
    entries.sort((a, b) => {
      // 按访问时间排序，最久未访问的在前
      return this.getLastAccessTime(a[0]) - this.getLastAccessTime(b[0])
    })

    let freedSize = 0
    for (const [url, localPath] of entries) {
      if (freedSize >= requiredSize) break

      const fileSize = await this.getImageSize(localPath)
      await this.deleteFile(localPath)
      this.imageCache.delete(url)
      this.currentCacheSize -= fileSize
      freedSize += fileSize
    }
  }
}
```

```typescript
class ErrorService {
  private errorBoundary: ErrorBoundary | null = null

      this.handleError(error, { isFatal, type: 'javascript' })
    })

    // 未处理的 Promise 拒绝
    const unhandledRejectionHandler = (reason: any) => {
      this.handleError(new Error(reason), { type: 'unhandled_promise' })
    }

    global.addEventListener?.('unhandledrejection', unhandledRejectionHandler)
  }

  handleError(error: Error, context?: ErrorContext): void {
    const errorInfo = {
      message: error.message,
      stack: error.stack,
      timestamp: new Date().toISOString(),
      context,
      deviceInfo: this.getDeviceInfo(),
      appState: this.getAppState(),
    }

    // 本地日志记录
    this.logErrorLocally(errorInfo)

    // 发送到服务器
    this.reportErrorToServer(errorInfo)

    // 用户通知
    if (context?.showToUser) {
      this.showErrorToUser(error)
    }
  }

  private async logErrorLocally(errorInfo: ErrorInfo): Promise<void> {
    try {
      const logEntry = JSON.stringify(errorInfo)
      await AsyncStorage.setItem(
        `error_log_${Date.now()}`,
        logEntry
      )

      // 清理旧的错误日志
      await this.cleanupOldLogs()
    } catch (logError) {
      console.error('Failed to log error:', logError)
    }
  }
}
```

```typescript
    const logEntry: LogEntry = {

      message,
      data,
      timestamp: new Date().toISOString(),
      sessionId: this.getSessionId(),
    }

    // 添加到缓冲区
    this.logBuffer.push(logEntry)

    // 控制台输出
    this.outputToConsole(logEntry)

    // 本地存储
    this.storeLogEntry(logEntry)

    // 缓冲区管理
    this.manageBuffer()
  }

  async flushLogs(): Promise<void> {
    if (this.logBuffer.length === 0) return

    try {
      const logs = [...this.logBuffer]
      await this.sendLogsToServer(logs)
      this.logBuffer = []
    } catch (error) {
      console.error('Failed to flush logs:', error)
    }
  }

    // 只在WiFi下发送日志，节省流量

      await api.post('/logs', { logs })
    } catch (error) {

      this.logBuffer.unshift(...logs)
    }
```
