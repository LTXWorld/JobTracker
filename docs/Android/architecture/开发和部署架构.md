#### 2.1.2 环境配置脚本

```bash
#!/bin/bash
# setup-dev-environment.sh

set -e

echo "🚀 Setting up JobView Android development environment"

# 检查系统依赖
check_system_deps() {
    echo "📋 Checking system dependencies..."

    # 检查 Node.js
    if ! command -v node &> /dev/null; then
        echo "❌ Node.js not found. Please install Node.js 20.10.0"
        exit 1
    fi

    NODE_VERSION=$(node --version | cut -d 'v' -f 2)
    if [[ ! "$NODE_VERSION" =~ ^20\. ]]; then
        echo "⚠️  Node.js version $NODE_VERSION found. Recommended: 20.10.0"
    fi

    # 检查 Java
    if ! command -v java &> /dev/null; then
        echo "❌ Java not found. Please install OpenJDK 17"
        exit 1
    fi

    # 检查 Android SDK
    if [ -z "$ANDROID_HOME" ]; then
        echo "❌ ANDROID_HOME not set. Please set Android SDK path"
        exit 1
    fi

    echo "✅ System dependencies check passed"
}

# 安装项目依赖
install_dependencies() {
    echo "📦 Installing project dependencies..."

    # 清理缓存
    npm cache clean --force
    rm -rf node_modules package-lock.json

    # 安装依赖
    npm install

    # React Native 依赖
    cd android && ./gradlew clean && cd ..

    # iOS 依赖 (if needed)
    if [ -d "ios" ]; then
        cd ios && pod install && cd ..
    fi

    echo "✅ Dependencies installed successfully"
}

# 配置开发工具
setup_dev_tools() {
    echo "🛠️  Setting up development tools..."

    # 安装全局工具
    npm install -g @react-native-community/cli@13.6.0
    npm install -g flipper@0.212.0
    npm install -g reactotron-cli@2.0.0

    # 配置 Git hooks
    npm run prepare

    # 配置 VS Code
    if [ -d ".vscode" ]; then
        echo "📝 VS Code configuration found"
    else
        mkdir -p .vscode
        cat > .vscode/settings.json << 'EOF'
{
  "typescript.preferences.importModuleSpecifier": "relative",
  "eslint.autoFixOnSave": true,
  "editor.formatOnSave": true,
  "editor.codeActionsOnSave": {
    "source.fixAll.eslint": true
  }
}
EOF
    fi

    echo "✅ Development tools configured"
}

# 验证环境
verify_environment() {
    echo "🔍 Verifying environment..."

    # React Native doctor
    npx react-native doctor

    # 运行基础测试
    npm run test -- --passWithNoTests

    echo "✅ Environment verification completed"
}

# 主执行流程
main() {
    check_system_deps
    install_dependencies
    setup_dev_tools
    verify_environment

    echo ""
    echo "🎉 Development environment setup completed!"
    echo ""
    echo "Next steps:"
    echo "  1. Run 'npm run android' to start the app"
    echo "  2. Open Android Studio for native development"
    echo "  3. Use 'npm run test' to run tests"
    echo ""
}

main "$@"
```

### 5.2 构建脚本

**scripts/build.sh**
```bash
#!/bin/bash

set -e

# 颜色定义
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
NC='\033[0m' # No Color

# 日志函数
log_info() {
    echo -e "${BLUE}[INFO]${NC} $1"
}

log_success() {
    echo -e "${GREEN}[SUCCESS]${NC} $1"
}

log_warning() {
    echo -e "${YELLOW}[WARNING]${NC} $1"
}

log_error() {
    echo -e "${RED}[ERROR]${NC} $1"
}

# 构建配置
BUILD_TYPE=${1:-debug}
VARIANT=${BUILD_TYPE^} # 首字母大写
OUTPUT_DIR="android/app/build/outputs"

log_info "Starting Android build process..."
log_info "Build type: $BUILD_TYPE"

# 清理前一次构建
clean_build() {
    log_info "Cleaning previous build..."
    cd android
    ./gradlew clean
    cd ..
    rm -rf node_modules/.cache
}

# 安装依赖
install_dependencies() {
    log_info "Installing dependencies..."
    npm ci

    if [ -d "android" ]; then
        cd android
        ./gradlew dependencies
        cd ..
    fi
}

# 运行测试
run_tests() {
    log_info "Running tests..."
    npm run lint
    npm run type-check
    npm run test -- --passWithNoTests
}

# 构建应用
build_app() {
    log_info "Building Android $BUILD_TYPE..."

    cd android

    if [ "$BUILD_TYPE" = "release" ]; then
        # 检查签名配置
        if [ ! -f "app/release.keystore" ]; then
            log_error "Release keystore not found!"
            exit 1
        fi

        if [ -z "$KEYSTORE_PASSWORD" ] || [ -z "$KEY_ALIAS" ] || [ -z "$KEY_PASSWORD" ]; then
            log_error "Signing configuration not complete!"
            exit 1
        fi

        # 构建签名版本
        ./gradlew assemble${Variant} bundle${Variant}
    else
        # 构建调试版本
        ./gradlew assemble${Variant}
    fi

    cd ..
}

# 验证构建产物
verify_build() {
    log_info "Verifying build artifacts..."

    if [ "$BUILD_TYPE" = "release" ]; then
        APK_PATH="$OUTPUT_DIR/apk/release/app-release.apk"
        AAB_PATH="$OUTPUT_DIR/bundle/release/app-release.aab"

        if [ -f "$APK_PATH" ]; then
            APK_SIZE=$(du -h "$APK_PATH" | cut -f1)
            log_success "APK built successfully: $APK_SIZE"
        else
            log_error "APK build failed!"
            exit 1
        fi

        if [ -f "$AAB_PATH" ]; then
            AAB_SIZE=$(du -h "$AAB_PATH" | cut -f1)
            log_success "AAB built successfully: $AAB_SIZE"
        else
            log_error "AAB build failed!"
            exit 1
        fi
    else
        APK_PATH="$OUTPUT_DIR/apk/debug/app-debug.apk"

        if [ -f "$APK_PATH" ]; then
            APK_SIZE=$(du -h "$APK_PATH" | cut -f1)
            log_success "Debug APK built successfully: $APK_SIZE"
        else
            log_error "Debug APK build failed!"
            exit 1
        fi
    fi
}

# 生成构建报告
generate_report() {
    log_info "Generating build report..."

    REPORT_FILE="build-report-$(date +%Y%m%d-%H%M%S).json"

    cat > "$REPORT_FILE" << EOF
{
  "buildTime": "$(date -u +%Y-%m-%dT%H:%M:%SZ)",
  "buildType": "$BUILD_TYPE",
  "gitCommit": "$(git rev-parse HEAD)",
  "gitBranch": "$(git rev-parse --abbrev-ref HEAD)",
  "artifacts": {
EOF

    if [ "$BUILD_TYPE" = "release" ]; then
        cat >> "$REPORT_FILE" << EOF
    "apk": {
      "path": "$APK_PATH",
      "size": "$(stat -f%z "$APK_PATH" 2>/dev/null || stat -c%s "$APK_PATH")"
    },
    "aab": {
      "path": "$AAB_PATH",
      "size": "$(stat -f%z "$AAB_PATH" 2>/dev/null || stat -c%s "$AAB_PATH")"
    }
EOF
    else
        cat >> "$REPORT_FILE" << EOF
    "apk": {
      "path": "$APK_PATH",
      "size": "$(stat -f%z "$APK_PATH" 2>/dev/null || stat -c%s "$APK_PATH")"
    }
EOF
    fi

    cat >> "$REPORT_FILE" << EOF
  }
}
EOF

    log_success "Build report generated: $REPORT_FILE"
}

# 主执行流程
main() {
    # 解析命令行参数
    while [[ $# -gt 0 ]]; do
        case $1 in
            --clean)
                CLEAN=true
                shift
                ;;
            --skip-tests)
                SKIP_TESTS=true
                shift
                ;;
            --skip-install)
                SKIP_INSTALL=true
                shift
                ;;
            *)
                BUILD_TYPE=$1
                shift
                ;;
        esac
    done

    # 验证构建类型
    if [[ "$BUILD_TYPE" != "debug" && "$BUILD_TYPE" != "release" ]]; then
        log_error "Invalid build type: $BUILD_TYPE"
        log_info "Usage: $0 [debug|release] [--clean] [--skip-tests] [--skip-install]"
        exit 1
    fi

    # 执行构建步骤
    if [ "$CLEAN" = true ]; then
        clean_build
    fi

    if [ "$SKIP_INSTALL" != true ]; then
        install_dependencies
    fi

    if [ "$SKIP_TESTS" != true ]; then
        run_tests
    fi

    build_app
    verify_build
    generate_report

    log_success "Build completed successfully!"
}

# 错误处理
trap 'log_error "Build failed on line $LINENO"' ERR

# 执行主函数
main "$@"
```

```bash
#!/bin/bash

set -e

# 部署配置
TRACK=${1:-internal}
SERVICE_ACCOUNT_KEY=${GOOGLE_PLAY_SERVICE_ACCOUNT_JSON}
PACKAGE_NAME="com.jobview.android"

log_info() {
    echo -e "\033[0;34m[INFO]\033[0m $1"
}

log_success() {
    echo -e "\033[0;32m[SUCCESS]\033[0m $1"
}

log_error() {
    echo -e "\033[0;31m[ERROR]\033[0m $1"
}

# 验证部署环境
validate_environment() {
    log_info "Validating deployment environment..."

    # 检查必要文件
    AAB_FILE="android/app/build/outputs/bundle/release/app-release.aab"
    if [ ! -f "$AAB_FILE" ]; then
        log_error "Release AAB not found: $AAB_FILE"
        exit 1
    fi

    # 检查服务账户密钥
    if [ -z "$SERVICE_ACCOUNT_KEY" ]; then
        log_error "Google Play service account key not provided"
        exit 1
    fi

    # 验证轨道参数
    if [[ "$TRACK" != "internal" && "$TRACK" != "alpha" && "$TRACK" != "beta" && "$TRACK" != "production" ]]; then
        log_error "Invalid track: $TRACK"
        log_info "Valid tracks: internal, alpha, beta, production"
        exit 1
    fi

    log_success "Environment validation passed"
}

# 上传到 Google Play Console
upload_to_play_console() {
    log_info "Uploading to Google Play Console (track: $TRACK)..."

    # 创建临时服务账户文件
    SERVICE_ACCOUNT_FILE=$(mktemp)
    echo "$SERVICE_ACCOUNT_KEY" > "$SERVICE_ACCOUNT_FILE"

    # 使用 fastlane 或 Google Play CLI 上传
    if command -v fastlane &> /dev/null; then
        log_info "Using Fastlane for deployment..."

        cat > Fastfile << EOF
default_platform(:android)

platform :android do
  desc "Deploy to Google Play Console"
  lane :deploy do
    upload_to_play_store(
      package_name: "$PACKAGE_NAME",
      aab: "$AAB_FILE",
      track: "$TRACK",
      json_key: "$SERVICE_ACCOUNT_FILE",
      skip_upload_metadata: true,
      skip_upload_images: true,
      skip_upload_screenshots: true
    )
  end
end
EOF

        fastlane android deploy

    else
        log_info "Using Google Play CLI for deployment..."

        # 安装 Google Play CLI (如果需要)
        if ! command -v androidpublisher &> /dev/null; then
            pip install google-api-python-client google-auth-httplib2 google-auth-oauthlib
        fi

        # Python 脚本上传
        python3 << EOF
import os
import json
from googleapiclient.discovery import build
from google.oauth2 import service_account

# 加载服务账户凭据
with open('$SERVICE_ACCOUNT_FILE', 'r') as f:
    service_account_info = json.load(f)

credentials = service_account.Credentials.from_service_account_info(
    service_account_info,
    scopes=['https://www.googleapis.com/auth/androidpublisher']
```

```typescript
// src/services/PerformanceMonitoringService.ts
import perf from '@react-native-firebase/perf'
import analytics from '@react-native-firebase/analytics'
import crashlytics from '@react-native-firebase/crashlytics'

export class PerformanceMonitoringService {
  private screenTraces = new Map<string, any>()

  // 应用启动性能
  async trackAppStart(): Promise<void> {
    const trace = await perf().startTrace('app_start')

    // 记录启动相关指标
    trace.putAttribute('cold_start', 'true')
    trace.putMetric('startup_time', Date.now())

    return trace
  }

  // 屏幕性能追踪
  async startScreenTrace(screenName: string): Promise<void> {
    if (this.screenTraces.has(screenName)) {
      return
    }

    const trace = await perf().startTrace(`screen_${screenName}`)
    trace.putAttribute('screen_name', screenName)

    this.screenTraces.set(screenName, {
      trace,
      startTime: Date.now(),
    })
  }

  async stopScreenTrace(screenName: string): Promise<void> {
    const traceData = this.screenTraces.get(screenName)
    if (!traceData) return

    const { trace, startTime } = traceData
    const duration = Date.now() - startTime

    trace.putMetric('screen_load_time', duration)
    trace.putMetric('memory_usage', await this.getMemoryUsage())

    await trace.stop()
    this.screenTraces.delete(screenName)
  }

  // API 请求性能
  async trackApiRequest(endpoint: string, method: string): Promise<any> {
    const trace = await perf().startTrace(`api_${method.toLowerCase()}_${endpoint.replace(/\//g, '_')}`)

    trace.putAttribute('endpoint', endpoint)
    trace.putAttribute('method', method)

    return {
      success: (responseTime: number, statusCode: number) => {
        trace.putMetric('response_time', responseTime)
        trace.putMetric('status_code', statusCode)
        trace.putAttribute('success', 'true')
        trace.stop()
      },
      error: (responseTime: number, errorCode: string) => {
        trace.putMetric('response_time', responseTime)
        trace.putAttribute('success', 'false')
        trace.putAttribute('error_code', errorCode)
        trace.stop()
      }
    }
  }

  // 自定义性能指标
  async trackCustomMetric(name: string, value: number, attributes?: Record<string, string>): Promise<void> {
    const trace = await perf().startTrace(name)

    if (attributes) {
      Object.entries(attributes).forEach(([key, val]) => {
        trace.putAttribute(key, val)
      })
    }

    trace.putMetric(name, value)
    await trace.stop()
  }

  // 内存使用监控
  private async getMemoryUsage(): Promise<number> {
    // React Native 中获取内存使用情况
    return new Promise((resolve) => {
      if (global.performance && global.performance.memory) {
        resolve(global.performance.memory.usedJSHeapSize)
      } else {
        resolve(0)
      }
    })
  }

  // 网络状态监控
  async trackNetworkPerformance(): Promise<void> {
    const trace = await perf().startTrace('network_performance')

    // 监控网络连接类型和速度
    import('@react-native-community/netinfo').then(NetInfo => {
      NetInfo.fetch().then(state => {
        trace.putAttribute('connection_type', state.type)
        trace.putAttribute('is_connected', state.isConnected.toString())

        if (state.details) {
          trace.putAttribute('is_wifi_enabled', state.details.isWifiEnabled?.toString() || 'unknown')
        }

        trace.stop()
      })
    })
  }

  // 用户行为分析
  async trackUserAction(action: string, screen: string, additionalData?: Record<string, any>): Promise<void> {
    await analytics().logEvent('user_action', {
      action,
      screen,
      timestamp: Date.now(),
      ...additionalData,
    })
  }

  // 应用使用时长
  private sessionStartTime = Date.now()

  trackSessionDuration(): void {
    const sessionDuration = Date.now() - this.sessionStartTime

    analytics().logEvent('session_duration', {
      duration: sessionDuration,
      session_start: this.sessionStartTime,
    })
  }
}

export const performanceMonitoring = new PerformanceMonitoringService()
```

```typescript
// src/services/ErrorTrackingService.ts
import crashlytics from '@react-native-firebase/crashlytics'

interface ErrorContext {
  screen?: string
  action?: string
  userId?: string
  additionalData?: Record<string, any>
}

export class ErrorTrackingService {
  // 初始化
  initialize(): void {
    // 在开发环境中禁用崩溃收集
    crashlytics().setCrashlyticsCollectionEnabled(!__DEV__)

    // 设置默认用户属性
    this.setDefaultAttributes()

    // 设置未捕获异常处理器
    this.setupGlobalErrorHandlers()
  }

  // 记录非致命错误
  recordError(error: Error, context?: ErrorContext): void {
    console.error('Recorded Error:', error, context)

    // 设置上下文
    if (context) {
      if (context.screen) {
        crashlytics().setAttribute('current_screen', context.screen)
      }
      if (context.action) {
        crashlytics().setAttribute('last_action', context.action)
      }
      if (context.additionalData) {
        Object.entries(context.additionalData).forEach(([key, value]) => {
          crashlytics().setAttribute(key, String(value))
        })
      }
    }

    // 记录错误
    crashlytics().recordError(error)
  }

  // 记录致命崩溃
  crash(message: string): void {
    crashlytics().crash()
  }

  // 设置用户信息
  setUser(userId: string, email?: string, name?: string): void {
    crashlytics().setUserId(userId)

    if (email) {
      crashlytics().setAttribute('user_email', email)
    }

    if (name) {
      crashlytics().setAttribute('user_name', name)
    }
  }

  // 设置自定义键
  setAttribute(key: string, value: string): void {
    crashlytics().setAttribute(key, value)
  }

  // 记录自定义日志
  log(message: string): void {
    crashlytics().log(message)
  }

  // 记录 API 错误
  recordApiError(endpoint: string, method: string, statusCode: number, error: any): void {
    const apiError = new Error(`API Error: ${method} ${endpoint}`)

    this.recordError(apiError, {
      additionalData: {
        endpoint,
        method,
        statusCode: statusCode.toString(),
        errorMessage: error.message || 'Unknown API error',
      }
    })
  }

  // 记录数据库错误
  recordDatabaseError(operation: string, table: string, error: any): void {
    const dbError = new Error(`Database Error: ${operation} on ${table}`)

    this.recordError(dbError, {
      additionalData: {
        operation,
        table,
        errorMessage: error.message || 'Unknown database error',
      }
    })
  }

  // 记录导航错误
  recordNavigationError(from: string, to: string, error: any): void {
    const navError = new Error(`Navigation Error: ${from} -> ${to}`)

    this.recordError(navError, {
      additionalData: {
        from_screen: from,
        to_screen: to,
        errorMessage: error.message || 'Unknown navigation error',
      }
    })
  }

  // 设置默认属性
  private setDefaultAttributes(): void {
    import('react-native-device-info').then(DeviceInfo => {
      // 设备信息
      DeviceInfo.getDeviceId().then(deviceId => {
        crashlytics().setAttribute('device_id', deviceId)
      })

      DeviceInfo.getSystemVersion().then(systemVersion => {
        crashlytics().setAttribute('system_version', systemVersion)
      })

      DeviceInfo.getAppVersion().then(appVersion => {
        crashlytics().setAttribute('app_version', appVersion)
      })

      DeviceInfo.getBuildNumber().then(buildNumber => {
        crashlytics().setAttribute('build_number', buildNumber)
      })
    })
  }

  // 设置全局错误处理器
  private setupGlobalErrorHandlers(): void {
    // React Native 错误边界
    const originalHandler = global.ErrorUtils.getGlobalHandler()

    global.ErrorUtils.setGlobalHandler((error, isFatal) => {
      console.error('Global Error:', error, 'isFatal:', isFatal)

      if (isFatal) {
        this.recordError(error, {
          additionalData: {
            is_fatal: 'true',
            stack_trace: error.stack || 'No stack trace',
          }
        })
      }

      // 调用原始处理器
      originalHandler(error, isFatal)
    })

    // Promise 拒绝处理
    global.addEventListener?.('unhandledrejection', (event) => {
      const error = new Error(`Unhandled Promise Rejection: ${event.reason}`)
      this.recordError(error, {
        additionalData: {
          rejection_reason: String(event.reason),
        }
      })
    })
  }

  // 测试崩溃报告
  testCrash(): void {
    if (__DEV__) {
      console.warn('Test crash called in development mode')
      return
    }

    crashlytics().crash()
  }
}

export const errorTracking = new ErrorTrackingService()
```
