### 3.2 主题配置系统

**src/theme/ThemeProvider.tsx**
```typescript
import React, { createContext, useContext, useEffect, useState } from 'react'
import { Appearance, ColorSchemeName } from 'react-native'
import { TamaguiProvider, Theme } from '@tamagui/core'
import AsyncStorage from '@react-native-async-storage/async-storage'
import { config } from './tamagui.config'

interface ThemeContextType {
  colorScheme: ColorSchemeName
  toggleTheme: () => void
  isSystemTheme: boolean
  setSystemTheme: (enabled: boolean) => void
}

const ThemeContext = createContext<ThemeContextType | undefined>(undefined)

export const useTheme = () => {
  const context = useContext(ThemeContext)
  if (!context) {
    throw new Error('useTheme must be used within ThemeProvider')
  }
  return context
}

interface ThemeProviderProps {
  children: React.ReactNode
}

export const ThemeProvider: React.FC<ThemeProviderProps> = ({ children }) => {
  const [colorScheme, setColorScheme] = useState<ColorSchemeName>('light')
  const [isSystemTheme, setIsSystemTheme] = useState(true)

  useEffect(() => {
    initializeTheme()
  }, [])

  useEffect(() => {
    if (isSystemTheme) {
      const subscription = Appearance.addChangeListener(({ colorScheme }) => {
        setColorScheme(colorScheme)
      })
      return () => subscription?.remove()
    }
  }, [isSystemTheme])

  const initializeTheme = async () => {
    try {
      const savedTheme = await AsyncStorage.getItem('color_scheme')
      const savedSystemTheme = await AsyncStorage.getItem('is_system_theme')

      if (savedSystemTheme === 'false') {
        setIsSystemTheme(false)
        setColorScheme(savedTheme as ColorSchemeName || 'light')
      } else {
        setIsSystemTheme(true)
        setColorScheme(Appearance.getColorScheme())
      }
    } catch (error) {
      console.error('Failed to load theme settings:', error)
    }
  }

  const toggleTheme = async () => {
    try {
      const newScheme = colorScheme === 'light' ? 'dark' : 'light'
      setColorScheme(newScheme)
      setIsSystemTheme(false)

      await AsyncStorage.multiSet([
        ['color_scheme', newScheme],
        ['is_system_theme', 'false'],
      ])
    } catch (error) {
      console.error('Failed to save theme settings:', error)
    }
  }

  const setSystemTheme = async (enabled: boolean) => {
    try {
      setIsSystemTheme(enabled)
      await AsyncStorage.setItem('is_system_theme', enabled.toString())

      if (enabled) {
        setColorScheme(Appearance.getColorScheme())
      }
    } catch (error) {
      console.error('Failed to save system theme setting:', error)
    }
  }

  const value: ThemeContextType = {
    colorScheme,
    toggleTheme,
    isSystemTheme,
    setSystemTheme,
  }

  return (
    <ThemeContext.Provider value={value}>
      <TamaguiProvider config={config}>
        <Theme name={colorScheme}>
          {children}
        </Theme>
      </TamaguiProvider>
    </ThemeContext.Provider>
  )
}
```

```typescript
export const applicationsApi = createApi({

          await queryFulfilled

      void,

      query: (updates) => ({
        url: 'status-tracking/batch',
        method: 'POST',
        body: { updates },
```

### 4.3 状态切片设计规范

**src/store/slices/applicationsSlice.ts**
```typescript
import { createSlice, createAsyncThunk, PayloadAction } from '@reduxjs/toolkit'
import { Application, ApplicationStatus, ApplicationFilters } from '@/types'
import { DatabaseService } from '@/services/DatabaseService'

interface ApplicationsState {
  items: Application[]
  filters: ApplicationFilters
  searchQuery: string
  selectedApplication: Application | null
  loading: boolean
  error: string | null
  lastSyncTime: string | null
  viewMode: 'kanban' | 'timeline' | 'list'
  sortBy: 'date' | 'company' | 'status'
  sortOrder: 'asc' | 'desc'
}

const initialState: ApplicationsState = {
  items: [],
  filters: {
    status: [],
    dateRange: null,
    company: '',
    position: '',
  },
  searchQuery: '',
  selectedApplication: null,
  loading: false,
  error: null,
  lastSyncTime: null,
  viewMode: 'kanban',
  sortBy: 'date',
  sortOrder: 'desc',
}

// 异步 Thunks
export const loadApplicationsFromDB = createAsyncThunk(
  'applications/loadFromDB',
  async (_, { rejectWithValue }) => {
    try {
      const dbService = new DatabaseService()
      const applications = await dbService.getAllApplications()
      return applications
    } catch (error) {
      return rejectWithValue('Failed to load applications from database')
    }
  }
)

export const syncApplications = createAsyncThunk(
  'applications/sync',
  async (_, { getState, rejectWithValue }) => {
    try {
      const state = getState() as { applications: ApplicationsState }
      const lastSync = state.applications.lastSyncTime

      // 调用 API 进行同步
      const syncService = new SyncService()
      const result = await syncService.performSync(lastSync)

      return result
    } catch (error) {
      return rejectWithValue('Sync failed')
    }
  }
)

const applicationsSlice = createSlice({
  name: 'applications',
  initialState,
  reducers: {
    setFilter: (state, action: PayloadAction<Partial<ApplicationFilters>>) => {
      state.filters = { ...state.filters, ...action.payload }
    },

    clearFilters: (state) => {
      state.filters = initialState.filters
      state.searchQuery = ''
    },

    setSearchQuery: (state, action: PayloadAction<string>) => {
      state.searchQuery = action.payload
    },

    setSelectedApplication: (state, action: PayloadAction<Application | null>) => {
      state.selectedApplication = action.payload
    },

    setViewMode: (state, action: PayloadAction<'kanban' | 'timeline' | 'list'>) => {
      state.viewMode = action.payload
    },

    setSorting: (
      state,
      action: PayloadAction<{ sortBy: string; sortOrder: 'asc' | 'desc' }>
    ) => {
      state.sortBy = action.payload.sortBy as any
      state.sortOrder = action.payload.sortOrder
    },

    // 本地状态更新（同步前）
    updateApplicationStatusLocally: (
      state,
      action: PayloadAction<{ id: number; status: ApplicationStatus }>
    ) => {
      const { id, status } = action.payload
      const application = state.items.find(app => app.id === id)
      if (application) {
        application.status = status
        application.updatedAt = new Date().toISOString()
      }
    },

    // 添加新应用到列表
    addApplicationLocally: (state, action: PayloadAction<Application>) => {
      state.items.unshift(action.payload)
    },

    // 从列表移除应用
    removeApplicationLocally: (state, action: PayloadAction<number>) => {
      state.items = state.items.filter(app => app.id !== action.payload)
    },
  },

  extraReducers: (builder) => {
    builder
      // 从数据库加载
      .addCase(loadApplicationsFromDB.pending, (state) => {
        state.loading = true
        state.error = null
      })
      .addCase(loadApplicationsFromDB.fulfilled, (state, action) => {
        state.loading = false
        state.items = action.payload
      })
      .addCase(loadApplicationsFromDB.rejected, (state, action) => {
        state.loading = false
        state.error = action.payload as string
      })

      // 同步
      .addCase(syncApplications.pending, (state) => {
        state.loading = true
        state.error = null
      })
      .addCase(syncApplications.fulfilled, (state, action) => {
        state.loading = false
        state.items = action.payload.applications
        state.lastSyncTime = new Date().toISOString()
      })
      .addCase(syncApplications.rejected, (state, action) => {
        state.loading = false
        state.error = action.payload as string
      })
  },
})

export const {
  setFilter,
  clearFilters,
  setSearchQuery,
  setSelectedApplication,
  setViewMode,
  setSorting,
  updateApplicationStatusLocally,
  addApplicationLocally,
  removeApplicationLocally,
} = applicationsSlice.actions

export default applicationsSlice.reducer

// Selectors
export const selectFilteredApplications = (state: { applications: ApplicationsState }) => {
  const { items, filters, searchQuery, sortBy, sortOrder } = state.applications

  let filtered = [...items]

  // 应用筛选
  if (filters.status.length > 0) {
    filtered = filtered.filter(app => filters.status.includes(app.status))
  }

  if (filters.company) {
    filtered = filtered.filter(app =>
      app.companyName.toLowerCase().includes(filters.company.toLowerCase())
    )
  }

  if (filters.position) {
    filtered = filtered.filter(app =>
      app.positionTitle.toLowerCase().includes(filters.position.toLowerCase())
    )
  }

  if (filters.dateRange) {
    filtered = filtered.filter(app => {
      const appDate = new Date(app.applicationDate)
      return appDate >= filters.dateRange!.start && appDate <= filters.dateRange!.end
    })
  }

  // 应用搜索
  if (searchQuery) {
    const query = searchQuery.toLowerCase()
    filtered = filtered.filter(app =>
      app.companyName.toLowerCase().includes(query) ||
      app.positionTitle.toLowerCase().includes(query) ||
      app.notes?.toLowerCase().includes(query)
    )
  }

  // 应用排序
  filtered.sort((a, b) => {
    let comparison = 0

    switch (sortBy) {
      case 'date':
        comparison = new Date(a.applicationDate).getTime() - new Date(b.applicationDate).getTime()
        break
      case 'company':
        comparison = a.companyName.localeCompare(b.companyName)
        break
      case 'status':
        comparison = a.status.localeCompare(b.status)
        break
    }

    return sortOrder === 'asc' ? comparison : -comparison
  })

  return filtered
}
```

### 5.1 Axios 配置

**src/services/ApiService.ts**
```typescript
import axios, { AxiosInstance, AxiosRequestConfig, AxiosResponse } from 'axios'
import AsyncStorage from '@react-native-async-storage/async-storage'
import NetInfo from '@react-native-netinfo/lib/commonjs'
import DeviceInfo from 'react-native-device-info'
import { Platform } from 'react-native'
import { Config } from '@/config'

interface ApiResponse<T = any> {
  success: boolean
  data: T
  message?: string
  errors?: string[]
}

interface RequestConfig extends AxiosRequestConfig {
  skipAuth?: boolean
  skipRetry?: boolean
  cacheKey?: string
  cacheTTL?: number
}

class ApiService {
  private client: AxiosInstance
  private responseCache = new Map<string, { data: any; expiry: number }>()
  private requestQueue: Array<() => Promise<any>> = []
  private isOnline = true

  constructor() {
    this.client = axios.create({
      baseURL: Config.API_BASE_URL,
      timeout: 30000,
      headers: {
        'Content-Type': 'application/json',
        'Accept': 'application/json',
        'X-App-Version': DeviceInfo.getVersion(),
        'X-Platform': Platform.OS,
        'X-Platform-Version': Platform.Version,
      },
    })

    this.setupInterceptors()
    this.setupNetworkMonitoring()
  }

  private setupInterceptors(): void {
    // 请求拦截器
    this.client.interceptors.request.use(
      async (config: RequestConfig) => {
        // 添加认证token
        if (!config.skipAuth) {
          const token = await this.getAuthToken()
          if (token) {
            config.headers.Authorization = `Bearer ${token}`
          }
        }

        // 检查缓存
        if (config.method === 'get' && config.cacheKey) {
          const cached = this.getFromCache(config.cacheKey)
          if (cached) {
            // 返回缓存的数据，跳过网络请求
            return Promise.reject({
              config,
              response: { data: cached },
              isCache: true,
            })
          }
        }

        // 添加请求签名
        const timestamp = Date.now().toString()
        config.headers['X-Timestamp'] = timestamp

        if (config.data) {
          const signature = await this.signRequest(config)
          config.headers['X-Signature'] = signature
        }

        return config
      },
      (error) => Promise.reject(error)
    )

    // 响应拦截器
    this.client.interceptors.response.use(
      (response: AxiosResponse) => {
        // 缓存响应
        const config = response.config as RequestConfig
        if (config.cacheKey && config.cacheTTL) {
          this.setCache(config.cacheKey, response.data, config.cacheTTL)
        }

        return response
      },
      async (error) => {
        // 处理缓存响应
        if (error.isCache) {
          return Promise.resolve(error.response)
        }

        const originalRequest = error.config as RequestConfig

        // 处理认证错误
        if (error.response?.status === 401 && !originalRequest.skipAuth) {
          try {
            const newToken = await this.refreshToken()
            if (newToken) {
              originalRequest.headers.Authorization = `Bearer ${newToken}`
              return this.client.request(originalRequest)
            }
          } catch (refreshError) {
            // Token刷新失败，跳转到登录页
            this.handleAuthError()
            return Promise.reject(refreshError)
          }
        }

        // 网络错误重试
        if (!originalRequest.skipRetry && this.shouldRetry(error)) {
          return this.retryRequest(originalRequest)
        }

        // 离线时添加到队列
        if (!this.isOnline && originalRequest.method !== 'get') {
          this.addToOfflineQueue(originalRequest)
        }

        return Promise.reject(error)
      }
    )
  }

  private async setupNetworkMonitoring(): Promise<void> {
    NetInfo.addEventListener(state => {
      const wasOnline = this.isOnline
      this.isOnline = state.isConnected ?? false

      // 网络恢复时处理离线队列
      if (!wasOnline && this.isOnline) {
        this.processOfflineQueue()
      }
    })
  }

  private async getAuthToken(): Promise<string | null> {
    try {
      const token = await AsyncStorage.getItem('auth_token')
      if (token && !this.isTokenExpired(token)) {
        return token
      }
      return await this.refreshToken()
    } catch (error) {
      console.error('Failed to get auth token:', error)
      return null
    }
  }

  private async refreshToken(): Promise<string | null> {
    try {
      const refreshToken = await AsyncStorage.getItem('refresh_token')
      if (!refreshToken) return null

      const response = await this.client.post('/auth/refresh',
        { refresh_token: refreshToken },
        { skipAuth: true, skipRetry: true }
      )

      const { token, refresh_token } = response.data.data

      await AsyncStorage.multiSet([
        ['auth_token', token],
        ['refresh_token', refresh_token],
      ])

      return token
    } catch (error) {
      console.error('Token refresh failed:', error)
      await this.clearTokens()
      return null
    }
  }

  private getFromCache(key: string): any | null {
    const cached = this.responseCache.get(key)
    if (cached && cached.expiry > Date.now()) {
      return cached.data
    }
    this.responseCache.delete(key)
    return null
  }

  private setCache(key: string, data: any, ttlMinutes: number): void {
    const expiry = Date.now() + (ttlMinutes * 60 * 1000)
    this.responseCache.set(key, { data, expiry })

    // 清理过期缓存
    this.cleanExpiredCache()
  }

  // 公共 API 方法
  async get<T = any>(
    url: string,
    config?: RequestConfig
  ): Promise<ApiResponse<T>> {
    const response = await this.client.get(url, config)
    return response.data
  }

  async post<T = any>(
    url: string,
    data?: any,
    config?: RequestConfig
  ): Promise<ApiResponse<T>> {
    const response = await this.client.post(url, data, config)
    return response.data
  }

  async put<T = any>(
    url: string,
    data?: any,
    config?: RequestConfig
  ): Promise<ApiResponse<T>> {
    const response = await this.client.put(url, data, config)
    return response.data
  }

  async delete<T = any>(
    url: string,
    config?: RequestConfig
  ): Promise<ApiResponse<T>> {
    const response = await this.client.delete(url, config)
    return response.data
  }

  // 批量请求
  async batch<T = any>(requests: Array<() => Promise<T>>): Promise<T[]> {
    const results = await Promise.allSettled(
      requests.map(request => request())
    )

    return results.map(result => {
      if (result.status === 'fulfilled') {
        return result.value
      } else {
        throw result.reason
      }
    })
  }
}
```

```typescript
import AsyncStorage from '@react-native-async-storage/async-storage'
import { LRUCache } from '@/utils/LRUCache'

interface CacheItem<T> {
  data: T
  timestamp: number
  ttl: number
  size?: number
}

interface CacheOptions {
  ttl?: number // 生存时间（分钟）
  maxSize?: number // 最大缓存大小（MB）
  strategy?: 'LRU' | 'TTL' | 'FIFO'
}

class CacheService {
  private memoryCache: LRUCache<string, any>
  private maxMemoryCacheSize = 50 // MB
  private currentMemorySize = 0

  constructor() {
    this.memoryCache = new LRUCache(1000) // 最多缓存1000个项目
  }

  // 内存缓存
  setMemoryCache<T>(key: string, data: T, options: CacheOptions = {}): void {
    const ttl = (options.ttl || 60) * 60 * 1000 // 转换为毫秒
    const item: CacheItem<T> = {
      data,
      timestamp: Date.now(),
      ttl,
      size: this.calculateSize(data),
    }

    // 检查内存使用量
    if (this.currentMemorySize + (item.size || 0) > this.maxMemoryCacheSize * 1024 * 1024) {
      this.cleanMemoryCache()
    }

    this.memoryCache.set(key, item)
    this.currentMemorySize += item.size || 0
  }

  getMemoryCache<T>(key: string): T | null {
    const item = this.memoryCache.get(key) as CacheItem<T> | undefined

    if (!item) return null

    // 检查是否过期
    if (Date.now() - item.timestamp > item.ttl) {
      this.memoryCache.delete(key)
      this.currentMemorySize -= item.size || 0
      return null
    }

    return item.data
  }

  // 持久化缓存
  async setPersistentCache<T>(
    key: string,
    data: T,
    options: CacheOptions = {}
  ): Promise<void> {
    try {
      const item: CacheItem<T> = {
        data,
        timestamp: Date.now(),
        ttl: (options.ttl || 60) * 60 * 1000,
      }

      await AsyncStorage.setItem(`cache_${key}`, JSON.stringify(item))
    } catch (error) {
      console.error('Failed to set persistent cache:', error)
    }
  }

  async getPersistentCache<T>(key: string): Promise<T | null> {
    try {
      const itemStr = await AsyncStorage.getItem(`cache_${key}`)
      if (!itemStr) return null

      const item: CacheItem<T> = JSON.parse(itemStr)

      // 检查是否过期
      if (Date.now() - item.timestamp > item.ttl) {
        await AsyncStorage.removeItem(`cache_${key}`)
        return null
      }

      return item.data
    } catch (error) {
      console.error('Failed to get persistent cache:', error)
      return null
    }
  }

  // 智能缓存（自动选择内存或持久化）
  async setSmartCache<T>(
    key: string,
    data: T,
    options: CacheOptions = {}
  ): Promise<void> {
    const size = this.calculateSize(data)
    const sizeInMB = size / (1024 * 1024)

    // 大于1MB的数据使用持久化缓存
    if (sizeInMB > 1) {
      await this.setPersistentCache(key, data, options)
    } else {
      this.setMemoryCache(key, data, options)
    }
  }

  async getSmartCache<T>(key: string): Promise<T | null> {
    // 先尝试内存缓存
    let result = this.getMemoryCache<T>(key)
    if (result) return result

    // 再尝试持久化缓存
    result = await this.getPersistentCache<T>(key)

    // 如果从持久化缓存中获取到数据，同时放入内存缓存
    if (result) {
      this.setMemoryCache(key, result, { ttl: 30 }) // 内存缓存30分钟
    }

    return result
  }

  // 清理缓存
  private cleanMemoryCache(): void {
    const entries = Array.from(this.memoryCache.entries())

    // 按访问时间排序，删除最少使用的
    entries.sort((a, b) => {
      const itemA = a[1] as CacheItem<any>
      const itemB = b[1] as CacheItem<any>
      return itemA.timestamp - itemB.timestamp
    })

    // 删除一半的缓存
    const toDelete = Math.floor(entries.length / 2)
    for (let i = 0; i < toDelete; i++) {
      const [key, item] = entries[i]
      this.memoryCache.delete(key)
      this.currentMemorySize -= (item as CacheItem<any>).size || 0
    }
  }

  async cleanExpiredPersistentCache(): Promise<void> {
    try {
      const keys = await AsyncStorage.getAllKeys()
      const cacheKeys = keys.filter(key => key.startsWith('cache_'))

      for (const key of cacheKeys) {
        const itemStr = await AsyncStorage.getItem(key)
        if (itemStr) {
          const item: CacheItem<any> = JSON.parse(itemStr)
          if (Date.now() - item.timestamp > item.ttl) {
            await AsyncStorage.removeItem(key)
          }
        }
      }
    } catch (error) {
      console.error('Failed to clean expired cache:', error)
    }
  }

  private calculateSize(data: any): number {
    return JSON.stringify(data).length * 2 // 粗略估算（UTF-16）
  }
}

export const cacheService = new CacheService()
```

```typescript
    applications: Application[],
    period: 'week' | 'month' | 'quarter' = 'month'
  ): ChartData[] {
    const now = new Date()
    const periodDays = period === 'week' ? 7 : period === 'month' ? 30 : 90
    const startDate = new Date(now.getTime() - periodDays * 24 * 60 * 60 * 1000)

    // 创建日期范围
    const dateRange: Date[] = []
    for (let i = 0; i < periodDays; i++) {
      const date = new Date(startDate)
      date.setDate(startDate.getDate() + i)
      dateRange.push(date)
    }

    // 统计每天的投递数量
    const dailyCounts = dateRange.map(date => {
      const dateStr = date.toISOString().split('T')[0]
      const count = applications.filter(app =>
        app.applicationDate.split('T')[0] === dateStr
      ).length

      return {
        x: date.getDate(),
        y: count,
        label: `${dateStr}: ${count} 个投递`,
      }
    })

    return dailyCounts
  }

  // 成功率分析柱状图数据
  static getSuccessRateAnalysis(applications: Application[]): ChartData[] {
    const stages = [
      '已投递',
      '简历筛选中',
      '笔试中',
      '一面中',
      '二面中',
      '三面中',
      'HR面中',
      '已收到offer',
    ]

    return stages.map(stage => {
      const stageApplications = applications.filter(app => {
        // 检查应用是否到达过这个阶段
        return this.hasReachedStage(app, stage)
      }).length

      const totalApplications = applications.length
      const rate = totalApplications > 0 ? (stageApplications / totalApplications) * 100 : 0

      return {
        x: stage,
        y: Math.round(rate),
        label: `${stage}: ${Math.round(rate)}%`,
      }
    })

    const companyCount = new Map<string, number>()

      const count = companyCount.get(app.companyName) || 0
      companyCount.set(app.companyName, count + 1)
    })

    // 按投递数量排序，取前N名
    const sortedCompanies = Array.from(companyCount.entries())

      '二面中',
      '三面中',
      'HR面中',
      '已收到offer',

    const targetIndex = stageOrder.indexOf(targetStage)

    if (data.length < windowSize) return data

    const smoothed: ChartData[] = []

    for (let i = 0; i < data.length; i++) {
      const start = Math.max(0, i - Math.floor(windowSize / 2))
      const end = Math.min(data.length, i + Math.ceil(windowSize / 2))

      const window = data.slice(start, end)
      const average = window.reduce((sum, item) => sum + item.y, 0) / window.length

      smoothed.push({
        ...data[i],
        y: Math.round(average * 100) / 100, // 保留2位小数
      })
    }

    return smoothed
  }
```

```typescript
import { InteractionManager, AppState, AppStateStatus } from 'react-native'

interface MemoryManagerOptions {
  maxMemoryUsage: number // MB
  cleanupInterval: number // 毫秒
  enableAutoCleanup: boolean
}

class MemoryManager {
  private options: MemoryManagerOptions
  private memoryUsage = 0
  private cleanupTimer: NodeJS.Timeout | null = null
  private isAppActive = true

  private imageCache = new Map<string, { data: string; lastAccessed: number; size: number }>()
  private componentCache = new Map<string, { component: React.ComponentType; lastAccessed: number }>()

  constructor(options: Partial<MemoryManagerOptions> = {}) {
    this.options = {
      maxMemoryUsage: 100, // 100MB
      cleanupInterval: 60000, // 1分钟
      enableAutoCleanup: true,
      ...options,
    }

    this.initialize()
  }

  private initialize(): void {
    // 监听应用状态变化
    AppState.addEventListener('change', this.handleAppStateChange)

    // 启动自动清理
    if (this.options.enableAutoCleanup) {
      this.startAutoCleanup()
    }
  }

  private handleAppStateChange = (nextAppState: AppStateStatus): void => {
    this.isAppActive = nextAppState === 'active'

    if (nextAppState === 'background') {
      // 应用进入后台时进行内存清理
      this.performEmergencyCleanup()
    }
  }

  // 图片缓存管理
  cacheImage(key: string, data: string, size?: number): void {
    const imageSize = size || this.estimateSize(data)

    // 检查内存使用量
    if (this.memoryUsage + imageSize > this.options.maxMemoryUsage * 1024 * 1024) {
      this.cleanupImageCache()
    }

    this.imageCache.set(key, {
      data,
      lastAccessed: Date.now(),
      size: imageSize,
    })

    this.memoryUsage += imageSize
  }

  getCachedImage(key: string): string | null {
    const cached = this.imageCache.get(key)
    if (cached) {
      cached.lastAccessed = Date.now()
      return cached.data
    }
    return null
  }

  // 组件缓存管理
  cacheComponent(key: string, component: React.ComponentType): void {
    this.componentCache.set(key, {
      component,
      lastAccessed: Date.now(),
    })
  }

  getCachedComponent(key: string): React.ComponentType | null {
    const cached = this.componentCache.get(key)
    if (cached) {
      cached.lastAccessed = Date.now()
      return cached.component
    }
    return null
  }

  // 清理方法
  private cleanupImageCache(): void {
    const entries = Array.from(this.imageCache.entries())

    // 按最后访问时间排序
    entries.sort(([, a], [, b]) => a.lastAccessed - b.lastAccessed)

    // 删除一半的缓存
    const toDelete = Math.floor(entries.length / 2)
    for (let i = 0; i < toDelete; i++) {
      const [key, cached] = entries[i]
      this.imageCache.delete(key)
      this.memoryUsage -= cached.size
    }
  }

  private cleanupComponentCache(): void {
    const entries = Array.from(this.componentCache.entries())
    const threshold = Date.now() - (30 * 60 * 1000) // 30分钟前

    entries.forEach(([key, cached]) => {
      if (cached.lastAccessed < threshold) {
        this.componentCache.delete(key)
      }
    })
  }

  private performEmergencyCleanup(): void {
    // 清理过期缓存
    this.cleanupImageCache()
    this.cleanupComponentCache()

    // 强制垃圾回收（如果可用）
    if (global.gc) {
      global.gc()
    }
  }

  private startAutoCleanup(): void {
    this.cleanupTimer = setInterval(() => {
      if (this.isAppActive) {
        // 应用活跃时轻度清理
        this.cleanupComponentCache()
      } else {
        // 应用非活跃时深度清理
        this.performEmergencyCleanup()
      }
    }, this.options.cleanupInterval)
  }

  private estimateSize(data: any): number {
    if (typeof data === 'string') {
      return data.length * 2 // UTF-16
    }
    return JSON.stringify(data).length * 2
  }

  // 公共方法
  getMemoryUsage(): number {
    return this.memoryUsage
  }

  clearAllCaches(): void {
    this.imageCache.clear()
    this.componentCache.clear()
    this.memoryUsage = 0
  }

  destroy(): void {
    if (this.cleanupTimer) {
      clearInterval(this.cleanupTimer)
    }
    AppState.removeEventListener('change', this.handleAppStateChange)
    this.clearAllCaches()
  }
}

export const memoryManager = new MemoryManager()
```

```typescript
import React, { Component, ErrorInfo, ReactNode } from 'react'

import { crashReporter } from '@/services/CrashReporter'

interface Props {
  children: ReactNode
  fallback?: (error: Error, reset: () => void) => ReactNode
}

interface State {
  hasError: boolean
  error: Error | null
  errorInfo: ErrorInfo | null
}

export class ErrorBoundary extends Component<Props, State> {
  constructor(props: Props) {
    super(props)
    this.state = { hasError: false, error: null, errorInfo: null }
  }

  static getDerivedStateFromError(error: Error): State {
    return { hasError: true, error, errorInfo: null }
  }

  componentDidCatch(error: Error, errorInfo: ErrorInfo): void {
    this.setState({ errorInfo })

    // 记录错误日志
    logger.error('React Error Boundary caught an error', error, {
      errorInfo,
      stack: error.stack,
    })

    // 上报崩溃信息
    crashReporter.recordError(error, {
      type: 'react_error_boundary',
      errorInfo,
    })
  }

  handleReset = (): void => {
    this.setState({ hasError: false, error: null, errorInfo: null })
  }

  render(): ReactNode {
    if (this.state.hasError && this.state.error) {
      if (this.props.fallback) {
        return this.props.fallback(this.state.error, this.handleReset)
      }

      return <DefaultErrorFallback error={this.state.error} onReset={this.handleReset} />

    return this.props.children
  }
```

```typescript
import AsyncStorage from '@react-native-async-storage/async-storage'
import { logger } from './LoggingService'

class DevTools {
  private isEnabled = __DEV__

  async initialize(): Promise<void> {
    if (!this.isEnabled) return

    // 启用网络调试
    if (global.XMLHttpRequest) {
      this.setupNetworkInterceptor()
    }

    // 启用异常调试
    this.setupErrorInterceptor()

    // 启用性能监控
    this.setupPerformanceMonitoring()
  }

  private setupNetworkInterceptor(): void {
    const originalXHR = global.XMLHttpRequest

    global.XMLHttpRequest = class extends originalXHR {
      open(...args: any[]): void {
        const [method, url] = args
        logger.debug(`Network Request: ${method} ${url}`)
        return super.open(...args)
      }
    }
  }

  private setupErrorInterceptor(): void {
    const originalConsoleError = console.error

    console.error = (...args: any[]) => {
      logger.error('Console Error', new Error(args.join(' ')))
      return originalConsoleError.apply(console, args)
    }
  }

  private setupPerformanceMonitoring(): void {
    // 监控应用启动时间
    const startTime = Date.now()

    global.addEventListener?.('load', () => {
      const loadTime = Date.now() - startTime
      logger.info(`App Load Time: ${loadTime}ms`)
    })

    // 监控内存使用
    setInterval(() => {
      if (global.performance?.memory) {
        const memory = global.performance.memory
        logger.debug('Memory Usage', {
          used: Math.round(memory.usedJSHeapSize / 1024 / 1024) + 'MB',
          total: Math.round(memory.totalJSHeapSize / 1024 / 1024) + 'MB',
        })
      }
    }, 30000) // 每30秒记录一次
  }

  // 调试辅助方法
  async clearAllStorage(): Promise<void> {
    try {
      await AsyncStorage.clear()
      logger.info('All AsyncStorage cleared')
    } catch (error) {
      logger.error('Failed to clear AsyncStorage', error as Error)
    }
  }

  async dumpStorageKeys(): Promise<void> {
    try {
      const keys = await AsyncStorage.getAllKeys()
      logger.debug('AsyncStorage Keys', { keys })
    } catch (error) {
      logger.error('Failed to get storage keys', error as Error)
    }
  }

  logReduxState(state: any): void {
    logger.debug('Redux State', state)
  }
}

export const devTools = new DevTools()

// 在开发环境下暴露到全局
if (__DEV__) {
  ;(global as any).devTools = devTools
}
```
