# JobView Android 数据库设计规范

## 1. 执行概要

本文档定义了 JobView Android 应用的完整数据库设计规范，包括 WatermelonDB 本地数据库模型设计、与 PostgreSQL 的数据同步策略、离线存储优化方案、数据冲突解决机制和数据安全备份策略。

## 2. 数据库架构概览

### 2.1 数据存储层次结构

```typescript
import { Model, field, date, text, readonly, writer, children, Q } from '@nozbe/watermelondb'
import { Associations } from '@nozbe/watermelondb/Model'

export default class Application extends Model {
  static table = 'applications'
  static associations: Associations = {
    statusHistories: { type: 'has_many', foreignKey: 'application_id' },
    reminders: { type: 'has_many', foreignKey: 'application_id' },
  }

  // 基本信息
  @text('company_name') companyName!: string
  @text('position_title') positionTitle!: string
  @date('application_date') applicationDate!: Date
  @text('status') status!: ApplicationStatus
  @text('salary_range') salaryRange?: string
  @text('work_location') workLocation?: string
  @text('notes') notes?: string
  @text('contact_info') contactInfo?: string
  @text('job_description') jobDescription?: string
  @text('requirements') requirements?: string

  // 元数据
  @text('user_id') userId!: string
  @date('created_at') createdAt!: Date
  @date('updated_at') updatedAt!: Date

  // 服务器同步字段
  @readonly @date('server_created_at') serverCreatedAt?: Date
  @readonly @date('server_updated_at') serverUpdatedAt?: Date
  @text('server_id') serverId?: string
  @text('sync_status') syncStatus!: SyncStatus // 'synced' | 'pending' | 'conflict' | 'deleted'
  @text('sync_version') syncVersion!: string
  @text('checksum') checksum!: string

  // 关联数据
  @children('status_histories') statusHistories!: Query<StatusHistory>
  @children('reminders') reminders!: Query<Reminder>

  // 业务方法
  @writer async updateStatus(
    newStatus: ApplicationStatus,
    note?: string,
    userId?: string
  ): Promise<void> {
    await this.batch(
      this.prepareUpdate(application => {
        application.status = newStatus
        application.updatedAt = new Date()
        application.syncStatus = 'pending'
        application.syncVersion = this.generateSyncVersion()
        application.checksum = this.generateChecksum()
      }),
      this.collections.get<StatusHistory>('status_histories').prepareCreate(history => {
        history.applicationId = this.id
        history.fromStatus = this.status
        history.toStatus = newStatus
        history.note = note
        history.userId = userId || this.userId
        history.createdAt = new Date()
        history.syncStatus = 'pending'
      })
    )
  }

  @writer async softDelete(): Promise<void> {
    await this.update(application => {
      application.syncStatus = 'deleted'
      application.updatedAt = new Date()
      application.syncVersion = this.generateSyncVersion()
    })
  }

  // 辅助方法
  private generateSyncVersion(): string {
    return `${Date.now()}_${Math.random().toString(36).substr(2, 9)}`
  }

  private generateChecksum(): string {
    const data = {
      companyName: this.companyName,
      positionTitle: this.positionTitle,
      status: this.status,
      updatedAt: this.updatedAt?.toISOString(),
    }
    return btoa(JSON.stringify(data)).replace(/[^a-zA-Z0-9]/g, '').substr(0, 16)
  }

  // 搜索方法
  static searchByText(query: string) {
    return Q.or(
      Q.where('company_name', Q.like(`%${query}%`)),
      Q.where('position_title', Q.like(`%${query}%`)),
      Q.where('notes', Q.like(`%${query}%`))
    )
  }

  // 筛选方法
  static filterByStatus(statuses: ApplicationStatus[]) {
    return Q.where('status', Q.oneOf(statuses))
  }

  static filterByDateRange(startDate: Date, endDate: Date) {
    return Q.and(
      Q.where('application_date', Q.gte(startDate.getTime())),
      Q.where('application_date', Q.lte(endDate.getTime()))
    )
  }
}
```

```typescript
export default class StatusHistory extends Model {
  static table = 'status_histories'
  static associations: Associations = {
    application: { type: 'belongs_to', key: 'application_id' },
  }

  @text('application_id') applicationId!: string
  @text('from_status') fromStatus!: ApplicationStatus
  @text('to_status') toStatus!: ApplicationStatus
  @text('note') note?: string
  @date('created_at') createdAt!: Date
  @text('user_id') userId!: string

  // 同步字段
  @text('server_id') serverId?: string
  @text('sync_status') syncStatus!: SyncStatus
  @date('server_created_at') serverCreatedAt?: Date

  @relation('applications', 'application_id') application!: Relation<Application>

  // 获取状态持续时间
  async getDuration(): Promise<number | null> {
    const previousHistory = await this.collections
      .get<StatusHistory>('status_histories')
      .query(
        Q.where('application_id', this.applicationId),
        Q.where('to_status', this.fromStatus),
        Q.sortBy('created_at', Q.desc),
        Q.take(1)
      )
      .fetch()

    if (previousHistory.length > 0) {
      return this.createdAt.getTime() - previousHistory[0].createdAt.getTime()
    }

    return null
  }
}
```

```typescript
    )
  }
```

```typescript
export default class SyncRecord extends Model {
  static table = 'sync_records'

  @text('entity_type') entityType!: string // 'application' | 'status_history' | 'reminder' | 'user'
  @text('entity_id') entityId!: string
  @text('operation') operation!: SyncOperation // 'create' | 'update' | 'delete'
  @text('sync_status') syncStatus!: SyncStatus
  @date('created_at') createdAt!: Date
  @date('synced_at') syncedAt?: Date
  @text('error_message') errorMessage?: string
  @text('retry_count') retryCount!: string
  @text('data_snapshot') dataSnapshot?: string // JSON字符串，用于冲突解决

  // 标记同步成功
  @writer async markSynced(): Promise<void> {
    await this.update(record => {
      record.syncStatus = 'synced'
      record.syncedAt = new Date()
    })
  }

  // 标记同步失败
  @writer async markFailed(errorMessage: string): Promise<void> {
    const retryCount = parseInt(this.retryCount) + 1
    await this.update(record => {
      record.syncStatus = 'pending'
      record.errorMessage = errorMessage
      record.retryCount = retryCount.toString()
    })
  }

  // 静态方法 - 获取待同步记录
  static getPendingSyncRecords(limit = 50) {
    return Q.and(
      Q.where('sync_status', 'pending'),
      Q.sortBy('created_at', Q.asc),
      Q.take(limit)
    )
  }
}
```

## 5. 数据同步策略

### 5.1 同步服务架构

```typescript
// src/services/SyncService.ts
import NetInfo from '@react-native-netinfo/lib/commonjs'
import { database } from '@/database'
import { apiService } from '@/services/ApiService'
import { logger } from '@/services/LoggingService'

interface SyncOptions {
  force?: boolean
  tables?: string[]
  batchSize?: number
  maxRetries?: number
}

interface SyncResult {
  success: boolean
  syncedRecords: number
  failedRecords: number
  conflicts: ConflictRecord[]
  errors: string[]
}

interface ConflictRecord {
  entityType: string
  entityId: string
  localData: any
  serverData: any
  conflictType: 'UPDATE_UPDATE' | 'UPDATE_DELETE' | 'DELETE_UPDATE'
}

export class SyncService {
  private isSyncing = false
  private syncQueue: SyncRecord[] = []
  private maxBatchSize = 50
  private maxRetries = 3

  async performFullSync(options: SyncOptions = {}): Promise<SyncResult> {
    if (this.isSyncing) {
      throw new Error('Sync already in progress')
    }

    try {
      this.isSyncing = true

      // 检查网络连接
      const netInfo = await NetInfo.fetch()
      if (!netInfo.isConnected) {
        throw new Error('No network connection')
      }

      logger.info('Starting full sync')

      // 1. 推送本地更改到服务器
      const pushResult = await this.pushLocalChanges(options)

      // 2. 从服务器拉取更新
      const pullResult = await this.pullServerChanges(options)

      // 3. 解决冲突
      const conflictResult = await this.resolveConflicts(
        [...pushResult.conflicts, ...pullResult.conflicts]
      )

      const result: SyncResult = {
        success: pushResult.success && pullResult.success,
        syncedRecords: pushResult.syncedRecords + pullResult.syncedRecords,
        failedRecords: pushResult.failedRecords + pullResult.failedRecords,
        conflicts: conflictResult.unresolvedConflicts,
        errors: [...pushResult.errors, ...pullResult.errors, ...conflictResult.errors],
      }

      logger.info('Full sync completed', result)
      return result

    } catch (error) {
      logger.error('Sync failed', error as Error)
      throw error
    } finally {
      this.isSyncing = false
    }
  }

  private async pushLocalChanges(options: SyncOptions): Promise<SyncResult> {
    try {
      // 获取待同步的本地记录
      const syncRecords = await database.collections
        .get<SyncRecord>('sync_records')
        .query(
          Q.where('sync_status', 'pending'),
          Q.sortBy('created_at', Q.asc),
          Q.take(options.batchSize || this.maxBatchSize)
        )
        .fetch()

      if (syncRecords.length === 0) {
        return {
          success: true,
          syncedRecords: 0,
          failedRecords: 0,
          conflicts: [],
          errors: [],
        }
      }

      // 按操作类型分组
      const operations = this.groupSyncRecordsByOperation(syncRecords)

      let syncedCount = 0
      let failedCount = 0
      const conflicts: ConflictRecord[] = []
      const errors: string[] = []

      // 处理创建操作
      if (operations.create.length > 0) {
        const createResult = await this.syncCreateOperations(operations.create)
        syncedCount += createResult.success
        failedCount += createResult.failed
        conflicts.push(...createResult.conflicts)
        errors.push(...createResult.errors)
      }

      // 处理更新操作
      if (operations.update.length > 0) {
        const updateResult = await this.syncUpdateOperations(operations.update)
        syncedCount += updateResult.success
        failedCount += updateResult.failed
        conflicts.push(...updateResult.conflicts)
        errors.push(...updateResult.errors)
      }

      // 处理删除操作
      if (operations.delete.length > 0) {
        const deleteResult = await this.syncDeleteOperations(operations.delete)
        syncedCount += deleteResult.success
        failedCount += deleteResult.failed
        conflicts.push(...deleteResult.conflicts)
        errors.push(...deleteResult.errors)
      }

      return {
        success: failedCount === 0,
        syncedRecords: syncedCount,
        failedRecords: failedCount,
        conflicts,
        errors,
      }

    } catch (error) {
      logger.error('Push local changes failed', error as Error)
      throw error
    }
  }

  private async pullServerChanges(options: SyncOptions): Promise<SyncResult> {
    try {
      // 获取最后同步时间
      const lastSyncTime = await this.getLastSyncTime()

      // 从服务器获取增量数据
      const response = await apiService.get('/sync/incremental', {
        params: {
          last_sync: lastSyncTime?.toISOString(),
          tables: options.tables?.join(','),
        },
      })

      const { applications, statusHistories, reminders, users, syncMetadata } = response.data

      let syncedCount = 0
      const conflicts: ConflictRecord[] = []
      const errors: string[] = []

      // 批量处理服务器数据
      await database.write(async () => {
        // 处理投递记录
        for (const serverApp of applications.created || []) {
          await this.createOrUpdateLocalRecord('applications', serverApp)
          syncedCount++
        }

        for (const serverApp of applications.updated || []) {
          const conflict = await this.updateLocalRecord('applications', serverApp)
          if (conflict) {
            conflicts.push(conflict)
          } else {
            syncedCount++
          }
        }

        for (const deletedId of applications.deleted || []) {
          await this.deleteLocalRecord('applications', deletedId)
          syncedCount++
        }

        // 处理其他表的数据...类似处理
      })

      // 更新同步时间
      await this.updateLastSyncTime(new Date(syncMetadata.current_timestamp))

      return {
        success: true,
        syncedRecords: syncedCount,
        failedRecords: 0,
        conflicts,
        errors,
      }

    } catch (error) {
      logger.error('Pull server changes failed', error as Error)
      throw error
    }
  }

  private async resolveConflicts(conflicts: ConflictRecord[]): Promise<{
    resolvedConflicts: number
    unresolvedConflicts: ConflictRecord[]
    errors: string[]
  }> {
    let resolvedCount = 0
    const unresolvedConflicts: ConflictRecord[] = []
    const errors: string[] = []

    for (const conflict of conflicts) {
      try {
        const resolved = await this.resolveConflict(conflict)
        if (resolved) {
          resolvedCount++
        } else {
          unresolvedConflicts.push(conflict)
        }
      } catch (error) {
        errors.push(`Failed to resolve conflict for ${conflict.entityType}:${conflict.entityId}`)
        unresolvedConflicts.push(conflict)
      }
    }

    return {
      resolvedConflicts: resolvedCount,
      unresolvedConflicts,
      errors,
    }
  }

  private async resolveConflict(conflict: ConflictRecord): Promise<boolean> {
    const { entityType, entityId, localData, serverData, conflictType } = conflict

    switch (conflictType) {
      case 'UPDATE_UPDATE':
        // 使用时间戳优先策略
        if (new Date(serverData.updated_at) > new Date(localData.updated_at)) {
          await this.applyServerData(entityType, serverData)
          return true
        } else {
          // 本地数据更新，推送到服务器
          await this.pushLocalData(entityType, localData)
          return true
        }

      case 'UPDATE_DELETE':
        // 服务器删除，本地更新 - 需要用户决定
        return false

      case 'DELETE_UPDATE':
        // 本地删除，服务器更新 - 需要用户决定
        return false

      default:
        return false
    }
  }
}
```

```typescript
export class IncrementalSyncService extends SyncService {
  private syncInterval: NodeJS.Timeout | null = null
  private isRealTimeEnabled = false

  async startRealTimeSync(): Promise<void> {
    if (this.isRealTimeEnabled) return

    this.isRealTimeEnabled = true

    // 监听网络状态变化
    NetInfo.addEventListener(state => {
      if (state.isConnected && this.syncQueue.length > 0) {
        this.processOfflineQueue()
      }
    })

    // 定期同步
    this.syncInterval = setInterval(() => {
      if (this.syncQueue.length > 0) {
        this.performIncrementalSync()
      }
    }, 30000) // 每30秒检查一次

    logger.info('Real-time sync started')
  }

  async stopRealTimeSync(): Promise<void> {
    this.isRealTimeEnabled = false

    if (this.syncInterval) {
      clearInterval(this.syncInterval)
      this.syncInterval = null
    }

    logger.info('Real-time sync stopped')
  }

  async queueLocalChange(
    entityType: string,
    entityId: string,
    operation: SyncOperation,
    data?: any
  ): Promise<void> {
    await database.write(async () => {
      const syncRecord = await database.collections
        .get<SyncRecord>('sync_records')
        .create(record => {
          record.entityType = entityType
          record.entityId = entityId
          record.operation = operation
          record.syncStatus = 'pending'
          record.createdAt = new Date()
          record.retryCount = '0'
          if (data) {
            record.dataSnapshot = JSON.stringify(data)
          }
        })

      this.syncQueue.push(syncRecord)
    })

    // 如果实时同步启用且网络可用，立即尝试同步
    if (this.isRealTimeEnabled) {
      const netInfo = await NetInfo.fetch()
      if (netInfo.isConnected) {
        this.debouncedSync()
      }
    }
  }

  private debouncedSync = debounce(() => {
    this.performIncrementalSync()
  }, 2000) // 2秒内的多次变更合并为一次同步

  private async performIncrementalSync(): Promise<void> {
    if (this.isSyncing) return

    try {
      await this.performFullSync({ batchSize: 20 })
    } catch (error) {
      logger.warn('Incremental sync failed', error as Error)
    }
  }

  private async processOfflineQueue(): Promise<void> {
    logger.info('Processing offline queue')

    const pendingRecords = await database.collections
      .get<SyncRecord>('sync_records')
      .query(Q.where('sync_status', 'pending'))
      .fetch()

    this.syncQueue = pendingRecords
    await this.performIncrementalSync()
  }
}

// 防抖函数
function debounce(func: Function, delay: number) {
  let timeoutId: NodeJS.Timeout
  return function (...args: any[]) {
    clearTimeout(timeoutId)
    timeoutId = setTimeout(() => func.apply(null, args), delay)
  }
}
```

```typescript
    cutoffDate.setDate(cutoffDate.getDate() - retentionDays)

    let deletedCount = 0

    await database.write(async () => {
      // 清理过期的状态历史记录（保留最近的）
      const oldHistories = await database.collections
        .get<StatusHistory>('status_histories')
        .query(
          Q.where('created_at', Q.lt(cutoffDate.getTime())),
          Q.sortBy('created_at', Q.asc)
        )
        .fetch()

      // 为每个应用保留最近10条历史记录
      const groupedHistories = new Map<string, StatusHistory[]>()
      oldHistories.forEach(history => {
        const appId = history.applicationId
        if (!groupedHistories.has(appId)) {
          groupedHistories.set(appId, [])
        }
        groupedHistories.get(appId)!.push(history)
      })

      for (const [appId, histories] of groupedHistories) {
        // 保留最近10条，删除其余的
        if (histories.length > 10) {
          const toDelete = histories.slice(0, histories.length - 10)
          for (const history of toDelete) {
            await history.destroyPermanently()
            deletedCount++
          }
        }
      }

      // 清理已完成的同步记录
      const completedSyncRecords = await database.collections
        .get<SyncRecord>('sync_records')
        .query(
          Q.where('sync_status', 'synced'),
          Q.where('synced_at', Q.lt(cutoffDate.getTime()))
        )
        .fetch()

      for (const record of completedSyncRecords) {
        await record.destroyPermanently()
        deletedCount++
      }
    })

    logger.info(`Cleaned up ${deletedCount} old records`)
    return deletedCount
  }
```

```typescript
// src/services/DatabaseCacheService.ts
export class DatabaseCacheService {
  private queryCache = new Map<string, { data: any; timestamp: number; ttl: number }>()
  private maxCacheSize = 100
  private defaultTTL = 5 * 60 * 1000 // 5分钟

  // 缓存查询结果
  cacheQuery(queryKey: string, data: any, ttl = this.defaultTTL): void {
    // 如果缓存已满，删除最旧的条目
    if (this.queryCache.size >= this.maxCacheSize) {
      const oldestKey = Array.from(this.queryCache.entries())
        .sort(([, a], [, b]) => a.timestamp - b.timestamp)[0][0]
      this.queryCache.delete(oldestKey)
    }

    this.queryCache.set(queryKey, {
      data,
      timestamp: Date.now(),
      ttl,
    })
  }

  // 获取缓存的查询结果
  getCachedQuery<T>(queryKey: string): T | null {
    const cached = this.queryCache.get(queryKey)
    if (!cached) return null

    // 检查是否过期
    if (Date.now() - cached.timestamp > cached.ttl) {
      this.queryCache.delete(queryKey)
      return null
    }

    return cached.data
  }

  // 清除相关缓存
  invalidateCache(pattern: string): void {
    const keysToDelete = Array.from(this.queryCache.keys())
      .filter(key => key.includes(pattern))

    keysToDelete.forEach(key => this.queryCache.delete(key))
  }

  // 清除所有缓存
  clearCache(): void {
    this.queryCache.clear()
  }

  // 生成查询键
  generateQueryKey(tableName: string, query: any): string {
    const queryString = JSON.stringify(query)
    return `${tableName}_${btoa(queryString).slice(0, 16)}`
  }
}
```

```typescript
  }

      const backupData = JSON.parse(backupDataString)

      await database.write(async () => {
        // 清除现有数据
        await this.clearAllData()

            })

          for (const appData of backupData.applications) {
            await database.collections.get<Application>('applications').create(app => {
              Object.assign(app, appData)
            })
          }

          for (const historyData of backupData.statusHistories) {
            await database.collections.get<StatusHistory>('status_histories').create(history => {

            })
          }

            await database.collections.get<Reminder>('reminders').create(reminder => {
              Object.assign(reminder, reminderData)
            })

          }
        }
      }

      logger.error('Backup cleanup failed', error as Error)
    }
  }
```

```typescript
// src/services/DatabasePerformanceService.ts
export class DatabasePerformanceService {
  private queryMetrics = new Map<string, {
    totalTime: number
    callCount: number
    averageTime: number
    maxTime: number
    lastCalled: number
  }>()

  // 包装查询以监控性能
  async monitorQuery<T>(queryName: string, queryFn: () => Promise<T>): Promise<T> {
    const startTime = Date.now()

    try {
      const result = await queryFn()
      const endTime = Date.now()
      const duration = endTime - startTime

      this.recordQueryMetrics(queryName, duration)

      // 如果查询时间过长，记录警告
      if (duration > 1000) { // 超过1秒
        logger.warn(`Slow query detected: ${queryName} took ${duration}ms`)
      }

      return result

    } catch (error) {
      logger.error(`Query failed: ${queryName}`, error as Error)
      throw error
    }
  }

  private recordQueryMetrics(queryName: string, duration: number): void {
    const existing = this.queryMetrics.get(queryName)

    if (existing) {
      existing.totalTime += duration
      existing.callCount += 1
      existing.averageTime = existing.totalTime / existing.callCount
      existing.maxTime = Math.max(existing.maxTime, duration)
      existing.lastCalled = Date.now()
    } else {
      this.queryMetrics.set(queryName, {
        totalTime: duration,
        callCount: 1,
        averageTime: duration,
        maxTime: duration,
        lastCalled: Date.now(),
      })
    }
  }

  // 获取性能报告
  getPerformanceReport(): any {
    const report = {}

    this.queryMetrics.forEach((metrics, queryName) => {
      report[queryName] = {
        averageTime: Math.round(metrics.averageTime),
        maxTime: metrics.maxTime,
        callCount: metrics.callCount,
        totalTime: metrics.totalTime,
        lastCalled: new Date(metrics.lastCalled).toISOString(),
      }
    })

    return {
      timestamp: new Date().toISOString(),
      queries: report,
      recommendations: this.generateRecommendations(),
    }
  }

  private generateRecommendations(): string[] {
    const recommendations: string[] = []

    this.queryMetrics.forEach((metrics, queryName) => {
      if (metrics.averageTime > 500) {
        recommendations.push(`Consider optimizing ${queryName} - average time: ${Math.round(metrics.averageTime)}ms`)
      }

      if (metrics.callCount > 100 && metrics.averageTime > 100) {
        recommendations.push(`${queryName} is called frequently (${metrics.callCount} times) - consider caching`)
      }
    })

    return recommendations
  }

  // 重置监控数据
  resetMetrics(): void {
    this.queryMetrics.clear()
  }
}
```
